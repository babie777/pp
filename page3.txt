3. Employee details (one object)
#include <iostream>
using namespace std;
class Employee {
int empId;
string name;
float salary;
public:
void accept() {
cout << "Enter employee ID, name and salary: ";
cin >> empId >> name >> salary;
}
void display() {
cout << "\nEmployee ID: " << empId;
cout << "\nName: " << name;
cout << "\nSalary: " << salary << endl;
}
};
int main() {
Employee e;
e.accept();
e.display();
return 0;
}
