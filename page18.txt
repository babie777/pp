18. Function template to swap integers and floats
#include <iostream>
using namespace std;
template <class T>
void swapValues(T &a, T &b) {
T temp = a;
a = b;
b = temp;
}
int main() {
int x = 10, y = 20;
float p = 1.5, q = 2.5;
cout << "Before integer swap: " << x << " " << y << endl;
swapValues(x, y);
cout << "After integer swap: " << x << " " << y << endl;
cout << "\nBefore float swap: " << p << " " << q << endl;
swapValues(p, q);
cout << "After float swap: " << p << " " << q << endl;
return 0;
}
