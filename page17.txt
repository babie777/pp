17. Diamond problem and solution using virtual base class
#include <iostream>
using namespace std;
class A {
public:
int value;
A() {
value = 10;
}
};
class B : virtual public A { };
class C : virtual public A { };
class D : public B, public C { };
int main() {
D obj;
obj.value = 50;
cout << "Value = " << obj.value << endl;
return 0;
}
