22. Single inheritance: Person and Student
#include <iostream>
using namespace std;
class Person {
protected:
string name;
int age;
public:
void acceptPerson() {
cout << "Enter name and age: ";
cin >> name >> age;
}
};
class Student : public Person {
int rollNo;
public:
void accept() {
acceptPerson();
cout << "Enter roll number: ";
cin >> rollNo;
}
void display() {
cout << "\nName: " << name;
cout << "\nAge: " << age;
cout << "\nRoll Number: " << rollNo << endl;
}
};
int main() {
Student s;
s.accept();
s.display();
return 0;
}
